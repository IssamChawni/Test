####################################################################################################
FUNCTION Parametres()
DEFINE i,j,k INTEGER

-- Date comptable
   INITIALIZE r_Bknom.* TO NULL
   SELECT * INTO r_Bknom.* FROM Bknom WHERE age="99000" AND ctab="778" AND cacc="PARAM1"
   LET g_dco = DATE(r_Bknom.lib1[1,10])
   IF DATE(g_dco) IS NULL THEN
      LET g_msgbox = "DATE COMPTABLE INVALIDE [778,PARAM1]."
      CALL Write_Ftrace("E")
   END IF
   LET g_msgbox = "DATE COMPTABLE         : ",g_dco
   CALL Write_Ftrace("A")

-- Nom fichier etat des rejets
   SET ISOLATION TO DIRTY READ;
   INITIALIZE F_Rejet TO NULL
   SELECT TRIM(lib2) INTO F_Rejet FROM Bknom WHERE age="99000" AND ctab="097" AND cacc="R1688"
   IF LENGTH(F_Rejet CLIPPED) = 0 THEN
      LET g_msgbox = "NOM ETAT DES REJETS NON DECLARE [097,R1688]"
      CALL Write_Ftrace("E")
   END IF

-- Nom fichier archivé si fichier rejeté
   SET ISOLATION TO DIRTY READ;
   INITIALIZE F_Erreur TO NULL
   SELECT TRIM(lib2) INTO F_Erreur FROM Bknom WHERE age="99000" AND ctab="097" AND cacc="R1689"
   IF LENGTH(F_Erreur CLIPPED) = 0 THEN
      LET g_msgbox = "NOM FICHIER ARCHIVE (SI FIC. BFI EN ANOMALIE) NON DECLARE [097,R1689]"
      CALL Write_Ftrace("E")
   END IF

-- Nom fichier destiné à RDJ
   SET ISOLATION TO DIRTY READ;
   INITIALIZE F_Output TO NULL
   SELECT TRIM(lib2) INTO F_Output FROM Bknom WHERE age="99000" AND ctab="097" AND cacc="R1680"
   IF LENGTH(F_Output CLIPPED) = 0 THEN
      LET g_msgbox = "NOM FICHIER DESTINE A RDJ NON DECLARE [097,R1680]"
      CALL Write_Ftrace("E")
   END IF

-- Nom fichier etat liste des opérations
   SET ISOLATION TO DIRTY READ;
   INITIALIZE F_Liste TO NULL
   SELECT TRIM(lib2) INTO F_Liste FROM Bknom WHERE age="99000" AND ctab="097" AND cacc="R1681"
   IF LENGTH(F_Liste CLIPPED) = 0 THEN
      LET g_msgbox = "NOM ETAT (LISTE DES OPERATIONS) NON DECLARE [097,R1681]"
      CALL Write_Ftrace("E")
   END IF

-- Nom fichier provenant de BFI, et Nom fichier archivé
   INITIALIZE r_Bknom.* TO NULL
   SELECT * INTO r_Bknom.* FROM Bknom WHERE age="99000" AND ctab="470" AND cacc="SPTIT044"
   IF SQLCA.SQLCODE != 0 THEN
      LET g_msgbox = "CLE [470,SPTIT044] INEXISTANTE"
      CALL Write_Ftrace("E")
   END IF
   IF LENGTH(r_Bknom.lib1 CLIPPED) = 0 THEN
      LET g_msgbox = "NOM FICHIER PROVENANT DE BFI NON RENSEIGNE [470,SPTIT044,LIB1]"
      CALL Write_Ftrace("E")
   END IF
   IF LENGTH(r_Bknom.lib4 CLIPPED) = 0 THEN
      LET g_msgbox = "NOM FICHIER D'ARCHIVE DU FICHIER BFI NON RENSEIGNE [470,SPTIT044,lib4]"
      CALL Write_Ftrace("E")
   END IF
   LET F_Input  =r_Bknom.lib1
   LET F_Archive=r_Bknom.lib4

END FUNCTION
####################################################################################################
FUNCTION Explore_IN()
DEFINE i,j,k  INTEGER
DEFINE vcmd   CHAR(100)
DEFINE retour SMALLINT

   --DISPLAY "Fichier                : ",F_input CLIPPED
   LET vcmd = "test -f ",F_input CLIPPED
   CALL exec_cmd (vcmd CLIPPED) RETURNING retour
   IF retour!=0 THEN
      LET g_msgbox = "PAS DE FICHIER A TRAITER"
      CALL Write_Ftrace("E")
   END IF

   LET g_Lus = 0
   LET g_Ecr = 0

   IF Size_File(F_Input) = 0 THEN
      LET g_msgbox = "FICHIER VIDE"
	  CALL Get_FileVide()
	  
	  LET g_msgbox = "NOMBRE LUS             : ",g_Lus USING "<<<<<&" 
      CALL Write_Ftrace("A")
      LET g_msgbox = "NOMBRE ECRITS          : ",g_Ecr USING "<<<<<&" 
      CALL Write_Ftrace("A")
	  
      CALL Write_FTrace("F")
   END IF

   IF Ctrl_File() = -1 THEN
      LET g_msgbox = "FICHIER CONTIENT DES ANOMALIES"
      CALL Write_Ftrace("E")
   END IF

   CALL Get_File()

   LET g_msgbox = "NOMBRE LUS             : ",g_Lus USING "<<<<<&"
   CALL Write_Ftrace("A")
   LET g_msgbox = "NOMBRE ECRITS          : ",g_Ecr USING "<<<<<&"
   CALL Write_Ftrace("A")

END FUNCTION

      LET r_BFI.isin         = vrec[014,025]   -- CHAR(12),                     -- ISIN
      LET r_BFI.ctit         = vrec[026,030]   -- CHAR(05),                     -- Code Titre
      LET r_BFI.ltit         = vrec[031,070]   -- CHAR(40),                     -- libellé titre
      LET r_BFI.ftit         = vrec[071,071]   -- CHAR(01),                     -- Forme du Titre
      LET r_BFI.ttit         = vrec[072,091]   -- CHAR(20),                     -- TYPE Titre
      LET r_BFI.tope         = vrec[092,092]   -- CHAR(01),                     -- C=Confirmation,A=Annulation

      LET r_BFI.dep_tautva   = vrec[285,290]   -- CHAR(06),                     -- Taux de TVA

   SELECT a.*,"Compte inexistant                                                " err
     FROM Tinput a,Bknom b
    WHERE b.age="99000" AND b.ctab="68Y" AND a.cope=b.cacc AND b.mnt1=0

   INSERT INTO Terreur
   SELECT a.*,"Sens opération erronée ("||a.tope||")"
     FROM Tinput a
    WHERE nvl(a.tope,"") NOT IN ("C","A")

   INSERT INTO Terreur
   SELECT a.*,"Code opération non prévu ("||a.cope||")"
     FROM Tinput a
    WHERE a.cope NOT IN (SELECT b.cacc FROM Bknom b WHERE b.age="99000" AND b.ctab="68Y")

-- Contrôle combine type opération+code opération (C+COPE ou A+COPE)
   INSERT INTO Terreur
   SELECT a.*,"Incohérence sens et code opération (C)"
     FROM Tinput a
    WHERE a.tope = "C" AND a.cope IN (SELECT b.cacc FROM Bknom b WHERE b.age="99000" AND b.ctab="68Y" AND nvl(b.lib2,"")="N")

   INSERT INTO Terreur
   SELECT a.*,"Incohérence sens et code opération (A)"
     FROM Tinput a
    WHERE a.tope = "A" AND a.cope IN (SELECT b.cacc FROM Bknom b WHERE b.age="99000" AND b.ctab="68Y" AND nvl(b.lib3,"")="N")

   INSERT INTO Terreur
   SELECT a.*,"Quantité non numérique"
     FROM Tinput a
    WHERE a.qtte  NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]";

   INSERT INTO Terreur
   SELECT a.*,"Cours non numérique"
     FROM Tinput a
    WHERE a.cours NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]";

   INSERT INTO Terreur
   SELECT a.*,"Montant brut non numérique"
     FROM Tinput a
    WHERE a.mntbrut NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]";

-- Contrôles des montants et taux de commission bourse
   INSERT INTO Terreur
   SELECT a.*,"Commission de bourse non numérique"
     FROM Tinput a
    WHERE a.mch_moncom NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]";

   INSERT INTO Terreur

     FROM Tinput a
    WHERE a.mch_taucom NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9]";

   INSERT INTO Terreur
   SELECT a.*,"Montant TVA sur commission de bourse non numérique"
     FROM Tinput a
    WHERE a.mch_montva NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]";

   INSERT INTO Terreur
   SELECT a.*,"Taux de TVA sur commission de bourse non numérique"
     FROM Tinput a
    WHERE a.mch_tautva NOT MATCHES "[0-9][0-9][0-9][0-9][0-9][0-9]";

-- Contrôles des montants et taux de commission intermediaire
   INSERT INTO Terreur

FUNCTION Get_File()
DEFINE i       INTEGER
DEFINE v_Bknom RECORD LIKE Bknom.*

   START REPORT OUTPUT_RDJ TO F_Output
   START REPORT Report_Lst TO F_Liste

   DECLARE Enrichi CURSOR FOR
      SELECT * FROM Tinput a,Bknom b
       WHERE a.cope=b.cacc AND b.age="99000" AND b.ctab="68Y"
       ORDER BY a.sequence
   FOREACH Enrichi INTO r_BFI.*,i,v_Bknom.*
      INITIALIZE r_RDJ.* TO NULL
      LET g_Lus = g_Lus + 1
      CALL Select_Delta(r_BFI.age,r_BFI.dev,r_BFI.ncp)
      LET r_BFI.lope  = Bkhis_Lib()
      LET r_RDJ.libc  = Bkhisl_Lib()
      LET r_RDJ.cli   = r_Bkcom.cli
      LET r_RDJ.bque  = "022"
      LET r_RDJ.cvi   = r_Bkbqe.cville USING "&&&"
      LET r_RDJ.rib   = r_Bkcom.ribdec
      LET r_RDJ.lage  = r_Bk001_lib1
      IF r_Bkcli.lib<"11" THEN
         LET r_RDJ.nom=r_Bk036_lib1 CLIPPED
         LET r_RDJ.Nom= r_RDJ.nom CLIPPED," ",r_Bkcli.nom
         LET r_RDJ.Nom= r_RDJ.nom CLIPPED," ",r_Bkcli.pre
      ELSE
         LET r_RDJ.Nom=r_Bkcli.nom
      END IF
      IF  LENGTH(r_Bkcli.nom CLIPPED) = 0
      AND LENGTH(r_Bkcli.pre CLIPPED) = 0 THEN
         LET r_RDJ.Nom   = r_Bkcom.inti
      END IF
      LET r_RDJ.adr1  = r_Bkacp.adr1
      LET r_RDJ.adr2  = r_Bkacp.adr2
      LET r_RDJ.adr3  = r_Bkacp.adr3
      LET r_RDJ.adr3  = r_Bkacp.adr3
      LET r_RDJ.ville = r_Bkacp.ville
      LET r_RDJ.cpos  = r_Bkacp.cpos
      LET r_RDJ.cpay  = r_Bkacp.cpay
      LET r_RDJ.pays  = r_Bk040_lib1
      LET r_RDJ.natope= Nature_ope()
      LET r_RDJ.avis  = "N"
      IF v_Bknom.mnt2=1 THEN
         LET r_RDJ.avis  = "O"
      END IF
      OUTPUT TO REPORT OUTPUT_RDJ()
      OUTPUT TO REPORT Report_Lst()
      LET g_Ecr = g_Ecr + 1
   END FOREACH
   FINISH REPORT OUTPUT_RDJ
   FINISH REPORT Report_Lst

END FUNCTION

FUNCTION Select_Delta(vage,vdev,vncp)
DEFINE vage LIKE Bkcom.age
DEFINE vdev LIKE Bkcom.dev
DEFINE vncp LIKE Bkcom.ncp

   INITIALIZE r_Bkcom.*,r_Bkcli.*,r_Bkacp.*,r_Bkadcli.*,r_Bk001_lib1,r_Bk036_lib1 TO NULL
   IF LENGTH(vage CLIPPED)=0 THEN
      RETURN
   END IF

   LET vage = "00",vage
   LET vdev =  "0",vdev
   LET vncp =  "0",vncp

   EXECUTE IMMEDIATE "SET ISOLATION TO DIRTY READ"

   SELECT * INTO r_Bkcom.*,r_Bkbqe.*,r_Bkacp.*,r_Bkcli.*
     FROM Bkcom a,Bkbqe b,outer Bkacp c,outer Bkcli d
    WHERE (a.age=vage AND a.dev=vdev AND a.ncp=vncp)
      AND (a.age=b.guib AND b.age="00022" AND b.etab="00022")
      AND (a.age=c.age AND a.dev=c.dev AND a.ncp=c.ncp AND a.suf=c.suf AND c.typ="C")
      AND (a.cli=d.cli)

-- Adresse compte inexistante
   IF LENGTH(r_Bkacp.age CLIPPED) = 0 THEN
      IF LENGTH(r_Bkcli.cli CLIPPED) != 0 THEN
      -- Adresse client
         CALL Select_Bkadcli(r_Bkcli.cli,"D")
         IF SQLCA.SQLCODE = 0 THEN
            LET r_Bkacp.adr1  = r_Bkadcli.adr1
            LET r_Bkacp.adr2  = r_Bkadcli.adr2
            LET r_Bkacp.adr3  = r_Bkadcli.adr3
            LET r_Bkacp.ville = r_Bkadcli.ville
            LET r_Bkacp.cpay  = r_Bkadcli.cpay
            LET r_Bkacp.cpos  = r_Bkadcli.cpos
         END IF
      END IF
   END IF

-- Intitulé agence
   CALL Select_Bknom("001",r_Bkcom.age)   LET r_Bk001_lib1 = r_Bknom.lib1
-- Intitulé client
   CALL Select_Bknom("036",r_Bkcli.lib)   LET r_Bk036_lib1 = r_Bknom.lib1
-- Intitulé pays
   CALL Select_Bknom("040",r_Bkacp.cpay)  LET r_Bk040_lib1 = r_Bknom.lib1

END FUNCTION
####################################################################################################
FUNCTION Select_Bknom(vctab,vcacc)
DEFINE vctab LIKE Bknom.ctab
DEFINE vcacc LIKE Bknom.cacc

   INITIALIZE r_Bknom.* TO NULL
   SELECT * INTO r_Bknom.* FROM Bknom WHERE age="99000" AND ctab=vctab AND cacc=vcacc

END FUNCTION
####################################################################################################
FUNCTION Select_Bkadcli(vcli,vtyp)
DEFINE vcli LIKE Bkadcli.cli
DEFINE vtyp LIKE Bkadcli.typ

   INITIALIZE r_Bkadcli.* TO NULL
   SELECT * INTO r_Bkadcli.* FROM Bkadcli WHERE cli=vcli AND typ=vtyp

END FUNCTION

FUNCTION Get_PrgName()
DEFINE varg CHAR(256)
DEFINE i,j  INTEGER

   LET varg = ARG_VAL(0)
   LET j    = LENGTH(varg CLIPPED)
   FOR i = j TO 1 STEP -1
      IF varg[i,i]= "/" THEN EXIT FOR END IF
   END FOR
   LET g_prgname = varg[i+1,j]
   LET varg      = g_prgname CLIPPED
   LET g_prgname = ""
   LET j    = LENGTH(varg CLIPPED)
   FOR i = 1 TO j
      IF varg[i,i] = "." THEN EXIT FOR END IF
      LET g_prgname = g_prgname CLIPPED,varg[i,i]
   END FOR

   LET g_prgname = g_prgname CLIPPED

END FUNCTION

FUNCTION Write_Ftrace(v_etat)

   LET g_hhmmss = TIME
   LET g_ddmmyy = TODAY
   CASE v_etat
   WHEN "D"
      LET g_base = fgl_GETENV("DATABASE")
      DATABASE g_base
      IF NUM_ARGS() = 0 THEN                   ## recuperation du code utilisateur
         LET g_uti = "AUTO"                    ## a partir de la ligne de commande
      ELSE                                     ## si non fournie, on affecte XXXX
        LET g_uti = ARG_VAL(1)                 ##
      END IF                                   ##

      LET g_heurdeb = g_hhmmss[1,2],g_hhmmss[4,5],g_hhmmss[7,8]
      LET v_nomfile = "/tmp/trace",g_prgname CLIPPED
      LET g_hndlog = base.Channel.create()

      CALL g_hndlog.openfile(v_nomfile,"w")

      CALL g_hndlog.write(g_msgbox[1,70])

      LET g_msgbox = "DEBUT DU PROGRAMME ",g_prgname[1,24],"        ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "BASE DE DONNEES        : ",g_base
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      CALL Insert_Sgtrace(v_etat)
   WHEN "F"
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "FIN NORMALE DU PROGRAMME ",g_prgname[1,24],"  ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      CALL g_hndlog.CLOSE()
      CALL Insert_Sgtrace(v_etat)
      CALL MAJ_Trace_Global()
   WHEN "E"
      IF SQLCA.SQLCODE >= 0 THEN
         CALL Insert_Sgtrace(v_etat)
      END IF
      CALL g_hndlog.write(g_msgbox[1,70])

      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "FIN ANORMALE DU PROGRAMME ",g_prgname[1,24]," ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      CALL g_hndlog.CLOSE()
      CALL MAJ_Trace_Global()
   WHEN "W"
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
      CALL Insert_Sgtrace(v_etat)
   OTHERWISE
      CALL g_hndlog.write(g_msgbox[1,70])
      --DISPLAY g_msgbox[1,70]
   END CASE
   IF v_etat = "E" OR v_etat = "F" THEN
      IF v_etat = "E" THEN
         CALL Delete_File()
         CALL Save_File(0)
      	 DISPLAY "1"
         EXIT PROGRAM(1)
      ELSE
         CALL Save_File(1)
         --CALL Script()
      	 DISPLAY "0"
         EXIT PROGRAM(0)
      END IF
   END IF

END FUNCTION

   CALL hndLs.OPENPIPE(vLs CLIPPED,"r")
   CALL hndLs.SETDELIMITER("")
   WHILE hndLs.READ(vsize)
   END WHILE

   RETURN vsize

END FUNCTION
####################################################################################################
FUNCTION Save_File(Flag)
DEFINE Flag   INTEGER
DEFINE vcmd   STRING
DEFINE hhmmss CHAR(10)

   LET hhmmss = TIME

-- Archive si OK
   IF Flag=1 THEN
      LET vcmd  = "mv ",F_Input CLIPPED," ",F_Archive CLIPPED,"_",g_dco USING "YYYYMMDD","_",
                  hhmmss[1,2],hhmmss[4,5],hhmmss[7,8]," 2>/dev/null"
   ELSE
      LET vcmd  = "mv ",F_Input CLIPPED," ",F_Erreur CLIPPED,"_",g_dco USING "YYYYMMDD","_",
                  hhmmss[1,2],hhmmss[4,5],hhmmss[7,8]," 2>/dev/null"
   END IF

   run vcmd CLIPPED

END FUNCTION
####################################################################################################
FUNCTION Delete_File()
DEFINE vcmd STRING

   LET vcmd  = "rm ",F_output CLIPPED," 2>/dev/null"
   run vcmd CLIPPED
   LET vcmd  = "rm ",F_Liste CLIPPED," 2>/dev/null"
   run vcmd CLIPPED

END FUNCTION
