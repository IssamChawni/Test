FUNCTION Parametres()
DEFINE vcmd        CHAR(100)
DEFINE vpar        CHAR(100)
DEFINE vDir        CHAR(100)
DEFINE vchr        CHAR(3)
DEFINE i,j         INTEGER
DEFINE vKo         INTEGER
DEFINE vexiste     SMALLINT

   LET vKo = 0

   SET ISOLATION TO DIRTY READ;
-- Date comptable
   INITIALIZE r_Bknom.* TO NULL 
   INITIALIZE g_dco     TO NULL
   SELECT * INTO r_Bknom.* FROM Bknom WHERE age = "99000" AND ctab = "778" AND cacc = "PARAM1"
   IF SQLCA.SQLCODE = 100 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "DATE COMPTABLE NON DECLAREE DANS [778,PARAM1]."
      CALL Write_Ftrace("A")
   ELSE
      LET g_dco = DATE(r_Bknom.lib1[1,10])
      IF DATE(g_dco) IS NULL THEN
         LET vKo = vKo + 1
         LET g_msgbox = "DATE COMPTABLE INVALIDE [778,PARAM1]."
         CALL Write_Ftrace("A")
      END IF
   END IF

-- Date compatble à forcer
   INITIALIZE g_date TO NULL
   FOR i = 1 TO NUM_ARGS()
      LET vpar = ARG_VAL(i)
      LET vchr = vpar[1,3]
      LET j    = LENGTH(vpar CLIPPED)
      IF vchr = "-d:" THEN LET g_date = vpar[4,j] END IF
   END FOR
   IF LENGTH(g_date CLIPPED) != 0 THEN
      IF DATE(g_date) IS NULL THEN
         LET vKo = vKo + 1
         LET g_msgbox = "DATE COMPTABLE A FORCER INVALIDE"
      ELSE   
         LET g_msgbox = "FORCER DATE COMPTABLE A: ",g_date
      END IF
   ELSE
      LET g_date = TODAY USING "DD/MM/YYYY"
      LET g_msgbox = "DATE COMPTABLE FORCER A: ",g_date
   END IF
   CALL Write_Ftrace("A")

-- Repertoire des lots comptable + Repertoire d'archivage
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Dir_In,Dir_Arc TO NULL
   SELECT trim(lib1),trim(lib4) INTO Dir_In,Dir_Arc FROM Bknom 
    WHERE age = "99000" AND ctab = "470" AND cacc = "SPLOAD72"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "CLE [470,SPLOAD72] NON DECLAREE"
      CALL Write_Ftrace("A")
   ELSE   
      IF LENGTH(Dir_In CLIPPED) = 0 THEN
         LET vKo = vKo + 1
         LET g_msgbox = "REPERTOIRE DES LOTS COMPTABLE VIDE[470,SPLOAD72,lib1]"
         CALL Write_Ftrace("A")
      ELSE
         LET vDir = Exists_Dir(Dir_In CLIPPED)
         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Dir_In CLIPPED," REPERTOIRE INEXISTANT [470,SPLOAD72,lib1]"
            CALL Write_Ftrace("A")
         ELSE
            LET g_msgbox = "REPERTOIRE A EXPLORER  : ",Dir_In
            CALL Write_Ftrace("A")
         END IF
      END IF
      IF LENGTH(Dir_Arc CLIPPED) = 0 THEN
         LET vKo = vKo + 1
         LET g_msgbox = "REPERTOIRE D'ARCHIVE DES LOTS TRAITES VIDE [470,SPLOAD72,lib4]"
         CALL Write_Ftrace("A")
      ELSE   
         LET vDir = Exists_Dir(Dir_Arc CLIPPED)
         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Dir_Arc CLIPPED," REPERTOIRE INEXISTANT [470,SPLOAD72,lib4]"
            CALL Write_Ftrace("A")
         END IF
      END IF
   END IF

-- Repertoire d'archive des fichiers non traites
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Dir_Rej TO NULL
   SELECT trim(lib2) INTO Dir_Rej FROM Bknom WHERE age = "99000" AND ctab = "097" AND cacc = "R438"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "REPERTOIRE D'ARCHIVE DES LOTS REJETES NON DECLARE[097,R438]"
      CALL Write_Ftrace("A")
   ELSE
      IF LENGTH(Dir_Rej CLIPPED) = 0 THEN
         LET vKo = vKo + 1
         LET g_msgbox = "REPERTOIRE D'ARCHIVE DES LOTS REJETES VIDE [097,R438]"
         CALL Write_Ftrace("A")
      ELSE   
         LET vDir = Exists_Dir(Dir_Rej CLIPPED)
         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Dir_Rej CLIPPED," REPERTOIRE INEXISTANT [097,R438]"
            CALL Write_Ftrace("A")
         END IF
      END IF
   END IF

-- Fichier Etat recap des fichiers traités
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Etat_Ok TO NULL
   SELECT trim(lib2) INTO Etat_Ok FROM Bknom WHERE age = "99000" AND ctab = "097" AND cacc = "R889"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "REPERTOIRE ETAT RECAP DES LOTS TRAITES NON DECLARE[097,R889]"
      CALL Write_Ftrace("A")
   ELSE
      IF LENGTH(Etat_Ok CLIPPED) = 0 THEN
         LET vKo = vKo + 1
         LET g_msgbox = "REPERTOIRE ETAT RECAP DES LOTS TRAITES VIDE [097,R889]"
         CALL Write_Ftrace("A")
      ELSE
         LET vDir = Exists_Dir(Etat_Ok CLIPPED)
         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Etat_Ok CLIPPED," REPERTOIRE INEXISTANT [097,R889]"

         END IF

   END IF

-- Fichier Etat recap des fichiers non traités
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Etat_Ko TO NULL
   SELECT trim(lib2) INTO Etat_Ko FROM Bknom WHERE age = "99000" AND ctab = "097" AND cacc = "R898"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "REPERTOIRE ETAT RECAP DES LOTS NON TRAITES NON DECLARE[097,R898]"
      CALL Write_Ftrace("A")
   ELSE
      IF LENGTH(Etat_Ko CLIPPED) = 0 THEN
         LET vKo = vKo + 1
         LET g_msgbox = "REPERTOIRE ETAT RECAP DES LOTS NON TRAITES VIDE [097,R898]"
         CALL Write_Ftrace("A")
      ELSE

         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Etat_Ko CLIPPED," REPERTOIRE INEXISTANT [097,R898]"
            CALL Write_Ftrace("A")
         END IF
      END IF
      LET Etat_Ko = Etat_Ko CLIPPED,"rapport_",g_dco USING "YYYYMMDD","_",g_heurdeb,"_rej"
   END IF

-- Repertoire des etats des ecritures comptables
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Dir_Etat TO NULL
   SELECT trim(lib2) INTO Dir_Etat FROM Bknom WHERE age = "99000" AND ctab = "097" AND cacc = "R875"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "REPERTOIRE ETAT DES MVTS PAR LOT NON DECLARE[097,R875]"
      CALL Write_Ftrace("A")
   ELSE   
      IF LENGTH(Dir_Etat CLIPPED) = 0 THEN

         LET g_msgbox = "REPERTOIRE ETAT DES MVTS PAR LOT VIDE [097,R875]"
         CALL Write_Ftrace("A")

            LET g_msgbox = Dir_Etat CLIPPED," REPERTOIRE INEXISTANT [097,R875]"
            CALL Write_Ftrace("A")
         END IF
      END IF
   END IF

-- Repertoire des fichiers etats des rejets ou des anomalies
   SET ISOLATION TO DIRTY READ;
   INITIALIZE Dir_Err TO NULL
   SELECT trim(lib2) INTO Dir_Err FROM Bknom WHERE age = "99000" AND ctab = "097" AND cacc = "R876"
   IF SQLCA.SQLCODE != 0 THEN
      LET vKo = vKo + 1
      LET g_msgbox = "REPERTOIRE DES REJETS PAR LOT NON DECLARE[097,R876]"

         LET vDir = Exists_Dir(Dir_Err CLIPPED)
         IF vDir CLIPPED != "+1" THEN
            LET vKo = vKo + 1
            LET g_msgbox = Dir_Err CLIPPED," REPERTOIRE INEXISTANT [097,R876]"
            CALL Write_Ftrace("A")
         END IF
      END IF
   END IF

   IF vKo != 0 THEN
      INITIALIZE g_msgbox TO NULL
      CALL Write_Ftrace("E")
   END IF

-- Contrôle taille des identifiants---------------------------------------------------------------------
   LET vKo = 0
   SET ISOLATION TO DIRTY READ;
   DECLARE CurIden CURSOR FOR SELECT cacc,trim(lib3) FROM Bknom WHERE age = "99000" AND ctab = "460" AND lib5 = "O"
   FOREACH CurIden INTO r_Bk460.cacc,r_Bk460.lib3
      IF LENGTH(r_Bk460.lib3 CLIPPED) != 4 THEN
         LET vKo = -1
         LET g_msgbox = "Colonne representant IDEN doit être déclaré sur 4 positions [460,",r_Bk460.cacc CLIPPED,"]"

      END IF 
   END FOREACH
   SET ISOLATION TO DIRTY READ;
   DECLARE CurNden CURSOR FOR
      SELECT trim(lib3),count(*) FROM Bknom
       WHERE age = "99000" AND ctab = "460" AND lib5 = "O" GROUP BY lib3 HAVING count(*) > 1
   FOREACH CurNden INTO r_Bk460.lib3,i
      LET vKo = -1
      LET g_msgbox = "Colonne representant IDEN non unique [460,",r_Bk460.lib3 CLIPPED,"]"
      CALL Write_Ftrace("A")
   END FOREACH
   IF vko = -1 THEN
      CALL Write_Ftrace("E")
   END IF
   
-- Chargement table des agences
   FOR i = 1 TO 9000 INITIALIZE Tab001[i].* TO NULL END FOR
   DECLARE CurAge CURSOR FOR SELECT * FROM Bknom WHERE age = "99000" AND ctab = "001"
   FOREACH CurAge INTO r_Bknom.*
      IF r_Bknom.cacc < "00001" THEN CONTINUE FOREACH END IF

      LET Tab001[i].lib = r_Bknom.lib1
   END FOREACH
-- Chargement table des des devises

   FOREACH CurDev INTO r_Bknom.*
      IF r_Bknom.cacc < "001" THEN CONTINUE FOREACH END IF
      IF r_Bknom.cacc > "999" THEN CONTINUE FOREACH END IF
      LET i = r_Bknom.cacc
      LET Tab005[i].dev = r_Bknom.cacc
      LET Tab005[i].lib = r_Bknom.lib1
   END FOREACH

END FUNCTION
####################################################################################################
FUNCTION Exists_Dir(vDir)
DEFINE vDir        CHAR(100)
DEFINE vcmd        CHAR(100)
DEFINE vchr        CHAR(1)
DEFINE i           INTEGER
DEFINE vexiste     SMALLINT

   IF LENGTH(vDir CLIPPED) = 0 THEN RETURN "+1" CLIPPED END IF
   
   FOR i = LENGTH(vDir CLIPPED) TO 1 STEP -1
      LET vchr = vDir[i,i]
      IF  vchr = "/" THEN EXIT FOR END IF
   END FOR
   LET vDir = vDir[1,i]
   LET vcmd = "test -d ",vDir CLIPPED
   CALL exec_cmd (vcmd CLIPPED) RETURNING vexiste
   IF vexiste != 0 THEN RETURN vDir CLIPPED END IF

   RETURN "+1"

END FUNCTION
####################################################################################################
FUNCTION ExploreDir()
DEFINE vls         CHAR(200)
DEFINE vfile       CHAR(200)
DEFINE vexec       CHAR(200)
DEFINE vstat       SMALLINT
DEFINE vNbre       INTEGER

   LET vNbre = 0
   LET g_Ko  = 0
   LET g_Ok  = 0

   LET vls = "ls ",Dir_In CLIPPED
   LET g_hndLs = base.Channel.create()
   CALL g_hndLs.OPENPIPE(vls,"r")

   WHILE g_hndLs.READ(g_file)

      IF LENGTH(g_file CLIPPED) = 0 THEN CONTINUE WHILE END IF
      CALL Traite_File()
      LET vNbre = vNbre + 1
   END WHILE   
   CALL g_hndLs.CLOSE()

   IF g_Ok > 0 THEN  FINISH REPORT Recap_Ok END IF
   IF g_Ko > 0 THEN  FINISH REPORT Recap_Ko END IF

   LET g_msgbox = "NBRE FICHIERS LUS      : ",vNbre
   CALL Write_Ftrace("A")
   LET g_msgbox = "FICHIERS TRAITES       : ",g_Ok
   CALL Write_Ftrace("A")
   LET g_msgbox = "FICHIERS NON TRAITES   : ",g_Ko
   CALL Write_Ftrace("A")

END FUNCTION

   IF Etat_Mvts()     = -1 THEN                         RETURN END IF

END FUNCTION
####################################################################################################
FUNCTION Evaluate_File()
DEFINE i,j,k,l,m,n INTEGER
DEFINE vstat       SMALLINT
DEFINE chr1        CHAR(1)
DEFINE chr4        CHAR(4)
DEFINE vnom        CHAR(100) 
DEFINE vexec       CHAR(200)

-- Existence Fichier à traiter
   LET vexec = "test -f ",Dir_In CLIPPED,g_file CLIPPED
   LET vstat = exec_cmd (vexec CLIPPED)
   IF vstat != 0 THEN
      CALL Edit_Recap_Ko("INEXISTANT OU ENDOMMAGE")
      RETURN -1
   END IF

-- Nombre de lignes à traiter
   LET i = 0
   LET vexec = "wc -l ",Dir_In CLIPPED,g_file CLIPPED," | awk '{print $1}' 2>/dev/null"
   LET g_hndwc = base.Channel.create()
   CALL g_hndwc.OPENPIPE(vexec CLIPPED,"r")
   CALL g_hndwc.SETDELIMITER("")
   WHILE g_hndwc.READ([i])
   END WHILE
   CALL g_hndwc.CLOSE()

   IF i = 0 THEN
      CALL Edit_Recap_Ko("VIDE")
      RETURN -1
   END IF

-- Verification du nom de fichier à traiter
   LET vnom = g_file CLIPPED
   LET l = LENGTH(vnom CLIPPED)
-- Taille du nom au moins 7 caractères
   IF l < 8 THEN
      CALL Edit_Recap_Ko("TAILLE DU NOM DE FICHIER DOIT ETRE SUPERIEUR A 7")
      RETURN -1
   END IF
-- Présence du caractère "-" au moins 1 fois
   LET k = 0
   FOR i = 1 TO l
       LET chr1 = vnom[i,i]
       IF chr1 = "-" THEN LET k = k + 1 EXIT FOR END IF
   END FOR
   IF k = 0 THEN
      CALL Edit_Recap_Ko("LE NOM DU FICHIER DOIT CONTENIR LE - AU MOINS 1 FOIS")
      RETURN -1
   END IF
-- Detecter le point à partir de la fin et ignorer le reste
   WHILE TRUE
      LET k = 0
      FOR i = l TO 1 STEP -1
          LET chr1 = vnom[i,i]
          IF chr1 != "." THEN CONTINUE FOR END IF
          LET k    = i - 1
          LET vnom = vnom[1,k]
          LET l    = LENGTH(vnom CLIPPED)
          EXIT FOR
      END FOR
      IF k = 0 THEN EXIT WHILE END IF
   END WHILE
-- Detecter numéro de lot
   INITIALIZE g_nlot TO NULL
   LET l = LENGTH(vnom CLIPPED)
   FOR i = l TO 1 STEP -1
       LET chr1 = vnom[i,i]
       IF chr1 != "-" THEN CONTINUE FOR END IF
       LET i = i + 1
       LET g_nlot = vnom[i,l]
       EXIT FOR
   END FOR
   IF LENGTH(g_nlot CLIPPED) != 6 THEN
      CALL Edit_Recap_Ko("NO DE LOT DOIT ETRE SUR 6 CARACTERES")
      RETURN -1
   END IF
   FOR i = 1 TO 6
       LET chr1 = g_nlot[i,i]
       IF chr1 NOT MATCHES "[0-9]" THEN
          CALL Edit_Recap_Ko("NO DE LOT NON NUMERIQUE")
          RETURN -1
       END IF
   END FOR
-- Determiner nom module
   INITIALIZE g_cacc TO NULL   
   LET l = LENGTH(vnom CLIPPED)
   FOR i = 1 TO l
       LET chr1 = vnom[i,i]
       IF chr1 != "-" THEN CONTINUE FOR END IF
       LET i = i - 1
       LET g_cacc = vnom[1,i]
       EXIT FOR
   END FOR
   IF LENGTH(g_cacc CLIPPED) = 0 THEN
      CALL Edit_Recap_Ko("IMPOSSIBLE DE DETECTER NOM MODULE")
      RETURN -1
   END IF
   LET g_cacc = UPSHIFT(g_cacc CLIPPED)
-- Validation nom module
   SET ISOLATION TO DIRTY READ;
   SELECT * INTO r_Bk460.* FROM Bknom 
    WHERE age = "99000" AND ctab = "460" AND trim(upper(cacc)) = trim(g_cacc)
   CASE SQLCA.SQLCODE
   WHEN 0
      LET r_Bk460.lib5 = UPSHIFT(r_Bk460.lib5)
      LET r_Bk460.lib2 = UPSHIFT(r_Bk460.lib2)
      IF LENGTH(r_Bk460.lib5 CLIPPED) = 0 THEN LET r_Bk460.lib5 = "N" END IF
      IF LENGTH(r_Bk460.mnt1 CLIPPED) = 0 THEN LET r_Bk460.mnt1 = 0   END IF
      IF LENGTH(r_Bk460.tau1 CLIPPED) = 0 THEN LET r_Bk460.tau1 = 0   END IF
   WHEN 100
      CALL Edit_Recap_Ko("NOM MODULE NON PARAMETRE DANS [ctab=460]")
      RETURN -1
   OTHERWISE
      LET g_msgbox = "ECHEC SELECT Bknom.SQL: ",SQLCA.SQLCODE USING "-<<<<<","."
      CALL Edit_Recap_Ko(g_msgbox CLIPPED)
      RETURN -1
   END CASE
-- Code utilisateur
   IF LENGTH(r_Bk460.lib2 CLIPPED) = 0 THEN
      LET g_msgbox = "MODULE (",g_cacc CLIPPED,"): PARAMETRE MANQUANT (lib2=uti)"
      CALL Edit_Recap_Ko(g_msgbox CLIPPED)
      RETURN -1
   END IF

-- Verifier si lot déjà traité
   LET g_sqlprepar = 'SELECT FIRST 1 * FROM Bkmvr WHERE uti = ? AND npr = ?'
   PREPARE p1 FROM g_sqlprepar CLIPPED
   EXECUTE p1 INTO r_sgmvr.*  USING r_Bk460.lib2, g_nlot 
   
   IF SQLCA.SQLCODE = 0 THEN
      CALL Edit_Recap_Ko("LOT DEJA TRAITE")
      CALL Archiver_File_Ok()
      RETURN -1
   END IF
-- Code identifiant 
-- IF r_Bk460.lib5 = "O" AND LENGTH(r_Bk460.lib3 CLIPPED) = 0 THEN
--    LET g_Ko = g_Ko + 1
--    LET g_msgbox = "MODULE (",g_cacc CLIPPED,"): PARAMETRE MANQUANT (lib3=iden)"
--    OUTPUT TO REPORT RECAP_KO(g_msgbox CLIPPED)
--    RETURN -1
-- END IF

   RETURN 0

END FUNCTION

FUNCTION Ctrl_Data()
DEFINE vrow,vKo    INTEGER
DEFINE verr        CHAR(79)
DEFINE veve        CHAR(2)
DEFINE vnom        CHAR(200)
DEFINE i           INTEGER

   LET vKo   = 0
   LET vrow  = 0
   LET vnom  = Dir_In CLIPPED,g_file CLIPPED
   LET g_hndIn = base.Channel.create()
   CALL g_hndIn.OPENFILE(vnom,"r")

   CALL g_hndIn.SETDELIMITER("|")

   WHILE g_hndIn.READ([r_Lot.*,r_libx])

      LET vrow = vrow + 1 
      INITIALIZE verr TO NULL
      LET veve = r_Lot.eve[1,2]
   -- Agence
      IF LENGTH(r_Lot.age CLIPPED) != 5 THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "age[?]" ELSE LET verr = verr CLIPPED,", age[?]" END IF
      ELSE
         IF r_Lot.age CLIPPED < "00001" OR r_Lot.age CLIPPED > "90000" THEN
            IF LENGTH(verr CLIPPED) = 0 THEN
               LET verr = "age[",r_Lot.age CLIPPED,"?]" 
            ELSE
               LET verr = verr CLIPPED,",age[",r_Lot.age CLIPPED,"?]"
            END IF
         ELSE
            IF LENGTH(Tab001[r_Lot.age].age CLIPPED) = 0 THEN
               IF LENGTH(verr CLIPPED) = 0 THEN
                  LET verr = "age[",r_Lot.age CLIPPED,"?]" 
               ELSE
                  LET verr = verr CLIPPED,",age[",r_Lot.age CLIPPED,"?]"
               END IF
            END IF
         END IF
      END IF
   -- Devise
      IF LENGTH(r_Lot.dev CLIPPED) != 3 THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "dev[?]" ELSE LET verr = verr CLIPPED,", dev[?]" END IF
      ELSE
         IF r_Lot.dev CLIPPED != "000" THEN 
            IF LENGTH(Tab005[r_Lot.dev].dev CLIPPED) = 0 THEN
               IF LENGTH(verr CLIPPED) = 0 THEN
                  LET verr = "dev[",r_Lot.dev CLIPPED,"?]" 
               ELSE
                  LET verr = verr CLIPPED,",dev[",r_Lot.dev CLIPPED,"?]"
               END IF
            END IF
         END IF
      END IF
   -- Eve
      IF veve != "YY" AND veve != "ZZ" THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "eve<>[ZZ,YY]" ELSE LET verr = verr CLIPPED,", eve<>[ZZ,YY]" END IF
      END IF
   -- Sen
      IF r_Lot.sen != "C" AND r_Lot.sen != "D" THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "sen<>[C,D]" ELSE LET verr = verr CLIPPED,", sen<>[C,D]" END IF
      END IF
   -- Uti
      IF r_Lot.uti != r_Bk460.lib2 THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "uti<>[",r_Bk460.lib2 CLIPPED,"]" ELSE LET verr = verr CLIPPED,", uti<>[",r_Bk460.lib2 CLIPPED,"]" END IF
      END IF
   -- Npr
      IF r_Lot.npr CLIPPED != g_nlot CLIPPED THEN
         IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "npr<>[",g_nlot CLIPPED,"]" ELSE LET verr = verr CLIPPED,", npr<>[",g_nlot CLIPPED,"]" END IF
      END IF
   -- Dco
--    IF LENGTH(r_Lot.dco CLIPPED) != 0 THEN
--       IF DATE(r_Lot.dco CLIPPED) IS NULL THEN
--          IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "dco[",r_Lot.dco CLIPPED,"]" ELSE LET verr = verr CLIPPED,", dco[",r_Lot.dco CLIPPED,"]" END IF
--       END IF
--    ELSE
--       IF LENGTH(g_date CLIPPED) != 0 THEN
--          LET r_Lot.dco = DATE(g_date)  
--       ELSE   
--          IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "dco[?]" ELSE LET verr = verr CLIPPED,", dco[?]" END IF
--       END IF   
--    END IF
   -- Dva
      IF LENGTH(r_Lot.dva CLIPPED) != 0 THEN
         IF DATE(r_Lot.dva CLIPPED) IS NULL THEN
            IF LENGTH(verr CLIPPED) = 0 THEN LET verr = "dva[",r_Lot.dva CLIPPED,"?]" ELSE LET verr = verr CLIPPED,", dva[",r_Lot.dva CLIPPED,"?]" END IF
         END IF
      END IF
   -- Montant de la transaction
      IF LENGTH(r_Lot.mon CLIPPED) = 0 THEN LET r_Lot.mon = 0 END IF
      IF LENGTH(vErr CLIPPED) != 0 THEN
         LET vKo = vKo + 1
         CALL Edit_Etat_Err(vrow,verr)
      END IF
   END WHILE
   CALL g_hndIn.CLOSE()

   IF vKo > 0 THEN 
      FINISH REPORT Etat_Err
      LET g_msgbox = "CONTIENT DES ANOMALIES.VOIR ",File_Err CLIPPED
      CALL Edit_Recap_Ko(g_msgbox CLIPPED)
      RETURN -1
   END IF

   RETURN 0

END FUNCTION

                        EXIT FOR
                     END IF
                     LET r_libx = Fnc_Ltrim(r_libx[i+1,300])
                     EXIT FOR
                  END IF
               IF i=1 AND LENGTH(c CLIPPED) != 0 THEN 
               	
                     LET r_Bkmvrl.lib   = r_libx[1,60]
                      --DISPLAY "r_Bkmvrl.lib"  ,r_Bkmvrl.lib     ------------
                     LET vord           = vord + 1
                     LET r_Bkmvrl.iden  = r_sgmvr.iden
                      --DISPLAY "r_Bkmvrl.iden"  ,r_Bkmvrl.iden   ------------
                     LET r_Bkmvrl.noseq = r_sgmvr.noseq
                     --DISPLAY "r_Bkmvrl.noseq"  ,r_Bkmvrl.noseq  ------------
                     LET r_Bkmvrl.ord   = vord 
                     --DISPLAY "r_Bkmvrl.ord"  ,r_Bkmvrl.ord      ------------  
                     LET r_Bkmvrl.lang  = "001"
                     --DISPLAY "r_Bkmvrl.lang"  ,r_Bkmvrl.lang    ------------
                     INSERT INTO Tkmvrl VALUES (r_Bkmvrl.*)
                     --DISPLAY "$",r_Bkmvrl.*,"$"
                     --DISPLAY "================= insert ===========> ",TIME
                     IF SQLCA.SQLCODE < 0 THEN 
                        LET g_msgbox = "ECHEC INSERT Tkmvrl.SQL: ",SQLCA.SQLCODE USING "-<<<<<","."
                        CALL Edit_Etat_Err(g_lus,g_msgbox CLIPPED)
                        LET vKo = vKo + 1
                        EXIT FOR
                     END IF
                     LET r_libx = Fnc_Ltrim(r_libx[62,300])
                     EXIT FOR
               	
               END IF 
               END FOR
               
               IF SQLCA.SQLCODE < 0 THEN EXIT WHILE END IF
            END WHILE
         ELSE   
            INITIALIZE r_sgmvr.iden TO NULL
         END IF
      ELSE
         INITIALIZE r_sgmvr.iden TO NULL
      END IF
   -- Insérer mouvement comptable
      LET r_sgmvr.dco = DATE(g_date)
   -- Forcer CTA à (O)ui
      LET r_sgmvr.cta = "O"
   -- Suffixe à "  " si NULL
      IF LENGTH(r_sgmvr.suf CLIPPED) = 0 THEN LET r_sgmvr.suf = "  " END IF

      START REPORT Recap_Ok TO Etat_Ok   
   END IF
   OUTPUT TO REPORT RECAP_OK()
   
   WHENEVER ERROR CALL F_Erreur

   RETURN 0

FUNCTION Etat_Mvts()

-- Etat de controle
   LET File_Ecr = Dir_Etat CLIPPED,g_cacc CLIPPED,"-",g_nlot CLIPPED,"-controle.txt"
   START REPORT Etat_Ctrl TO File_Ecr
   DECLARE CurCtrl CURSOR FOR 
      SELECT dco,age,dev,sum(decode(sen,"D",mon,0)),sum(decode(sen,"C",mon,0))
        FROM Bkmvr
       WHERE uti = r_Bk460.lib2 AND npr = g_nlot 
       GROUP BY dco,age,dev
       ORDER BY age,dev,dco
   FOREACH CurCtrl INTO r_sgmvr.dco,r_sgmvr.age,r_sgmvr.dev,r_sgmvr.mon,r_sgmvr.mctv
      OUTPUT TO REPORT Etat_Ctrl(r_sgmvr.*)
      INITIALIZE r_sgmvr.* TO NULL
   END FOREACH
   FINISH REPORT Etat_Ctrl

   IF r_Bk460.mnt5 != 1 THEN RETURN 0 END IF

-- Etat des ecritures comptables
   LET File_Ecr = Dir_Etat CLIPPED,g_cacc CLIPPED,"-",g_nlot CLIPPED,".txt"
   START REPORT Etat_Ecr TO File_Ecr
   DECLARE CurMvts CURSOR FOR 
      SELECT * FROM Bkmvr WHERE uti = r_Bk460.lib2 AND npr = g_nlot ORDER BY age,dev,uti,rlet
   FOREACH CurMvts INTO r_Lot.*
      OUTPUT TO REPORT Etat_Ecr(r_Lot.*)
   END FOREACH
   FINISH REPORT Etat_Ecr

   IF LENGTH(r_Bk460.lib4 CLIPPED)!=0 THEN
      LET vrun="cp ",File_Ecr CLIPPED," ",r_Bk460.lib4 CLIPPED," 2>/dev/null"
      run vrun
   END IF

   RETURN 0

END FUNCTION

FUNCTION Fnc_Ltrim(vstr)
DEFINE vstr        CHAR(300)
DEFINE vchr        CHAR(1)
DEFINE i           INTEGER

   IF LENGTH(vstr CLIPPED) = 0 THEN RETURN vstr CLIPPED END IF
   FOR i = 1 TO LENGTH(vstr CLIPPED)
     LET vchr = vstr[i,i]
     IF vchr != " " THEN 
        LET vstr = vstr[i,300]
        EXIT FOR
     END IF
  END FOR

  RETURN vstr CLIPPED

END FUNCTION
####################################################################################################
FUNCTION Fnc_Replace(vstr)
DEFINE vstr        CHAR(300)
DEFINE vnew        CHAR(300)
DEFINE vchr        CHAR(1)
DEFINE vold        CHAR(1)
DEFINE i,j         INTEGER

   IF LENGTH(vstr CLIPPED) = 0 THEN RETURN vstr CLIPPED END IF

   LET vold = " "
   LET j    = 0
   FOR i = 1 TO LENGTH(vstr CLIPPED)
     LET vchr = vstr[i,i]
     IF vold = vchr AND vchr = " " THEN CONTINUE FOR END IF
     LET j = j + 1
     LET vnew[j,j] = vchr
     LET vold = vchr
  END FOR
  
  RETURN vnew CLIPPED

END FUNCTION

FUNCTION F_RollBack()
DEFINE CodeStat      INTEGER

  IF SQLCA.SQLCODE < 0 THEN
     LET g_msgbox = "Erreur Transaction SQL : ",SQLCA.SQLCODE
  ELSE
     LET CodeStat = status
     LET g_msgbox = "Erreur                 : ",CodeStat,". ",ERR_GET(CodeStat)
  END IF
  ROLLBACK WORK;
  CALL Write_FTrace("E")

END FUNCTION

   LET g_msgbox = "Erreur                 : ",CodeStat USING "-<<<<<",". ",ERR_GET(CodeStat)
   CALL Write_FTrace("E")

END FUNCTION

FUNCTION Write_Ftrace(v_etat)
DEFINE v_etat        CHAR(1)
DEFINE v_nomfile     CHAR(80)
Define Arg0          CHAR(256)
Define ix            SMALLINT

   LET g_hhmmss = TIME
   LET g_ddmmyy = TODAY
   CASE v_etat
   WHEN "D"
      LET g_base = fgl_GETENV("DATABASE")
      DATABASE g_base

      IF NUM_ARGS() = 0 THEN                   ## recuperation du code utilisateur
         LET g_uti = "XXXX"                    ## a partir de la ligne de commande
      ELSE                                     ## si non fournie, on affecte XXXX
        LET g_uti = ARG_VAL(1)                 ##
      END IF                                   ##

      LET ARG0 = ARG_VAL(0)
      FOR IX = LENGTH(ARG0) TO 1 STEP -1
         IF ARG0[IX,IX]= "/" THEN
            EXIT FOR
         END IF
      END FOR
      LET g_prgname = ARG0[IX+1,LENGTH(ARG0)]
      LET arg0      = g_prgname CLIPPED
      LET g_prgname = ""
      FOR ix = 1 TO LENGTH(arg0)
          IF arg0[ix,ix] = "." THEN EXIT FOR END IF
          LET g_prgname = g_prgname CLIPPED,arg0[ix,ix]
      END FOR
      LET g_prgname = g_prgname CLIPPED
      LET g_datedeb = g_ddmmyy
      LET g_heurdeb = g_hhmmss[1,2],g_hhmmss[4,5],g_hhmmss[7,8]
      LET v_nomfile = "/tmp/trace",g_prgname CLIPPED
      LET g_hndlog = base.Channel.create()
      CALL g_hndlog.setdelimiter("")
      CALL g_hndlog.openfile(v_nomfile,"w")

      
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "DEBUT DU PROGRAMME ",g_prgname[1,24],"        ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "BASE DE DONNEES        : ",g_base
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      CALL Insert_Sgtrace(v_etat)
   WHEN "F"
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "FIN NORMALE DU PROGRAMME ",g_prgname[1,24],"  ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      CALL g_hndlog.CLOSE()

      CALL Insert_Sgtrace(v_etat)
      CALL MAJ_Trace_Global()      
   WHEN "E"
      IF SQLCA.SQLCODE >= 0 THEN
         CALL Insert_Sgtrace(v_etat)
      END IF   
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "FIN ANORMALE DU PROGRAMME ",g_prgname[1,24]," ",g_ddmmyy," ",g_hhmmss
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      LET g_msgbox = "======================================================================"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      CALL g_hndlog.CLOSE()

      CALL MAJ_Trace_Global()      
   WHEN "W"
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
      CALL Insert_Sgtrace(v_etat)
   OTHERWISE
      CALL g_hndlog.WRITE(g_msgbox[1,70])

      DISPLAY g_msgbox[1,70]
   END CASE

   IF v_etat = "E" OR v_etat = "F" THEN
      IF v_etat = "E" THEN
         EXIT PROGRAM(1)
      ELSE
         CALL Script()
         EXIT PROGRAM(0)
      END IF      
   END IF

END FUNCTION

FUNCTION Insert_SgTrace(v_etat)
DEFINE v_etat        CHAR(1)
DEFINE v_heurfin     CHAR(6)
DEFINE v_datefin     DATE
DEFINE v_heure       CHAR(10)

   LET v_datefin = TODAY
   LET v_heure   = TIME
   LET v_heurfin = v_heure[1,2],v_heure[4,5],v_heure[7,8]
   CASE v_etat
   WHEN "D"
      LET g_msgbox = "DEBUT NORMAL"      
      LET v_datefin  = ""
      LET v_heurfin  = ""
   WHEN "F"
      LET g_msgbox = "FIN NORMALE"      
   END CASE
   INSERT INTO sgtrace VALUES
   (g_prgname,g_datedeb,g_heurdeb,v_datefin,v_heurfin,g_uti,v_etat,g_msgbox);

END FUNCTION

   WHENEVER ERROR CONTINUE

      run v_run
   END IF

END FUNCTION
